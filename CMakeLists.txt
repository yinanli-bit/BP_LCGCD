cmake_minimum_required(VERSION 3.20)

project(BP_LCGCD VERSION 0.1.0 LANGUAGES C)

add_executable(bp_lcgcd src/apps/main.c)
add_library(bplcgcd_gf2 STATIC src/core/gf2.c src/core/systematic.c)
target_include_directories(bplcgcd_gf2 PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
add_library(bplcgcd_decoder STATIC src/decoders/decoder.c)
target_link_libraries(bplcgcd_decoder PUBLIC bplcgcd_gf2 bplcgcd_search)
add_library(bplcgcd_search STATIC src/search/tep.c src/search/local_constraints.c src/search/trellis.c src/search/viterbi.c src/search/slva.c)
target_link_libraries(bplcgcd_search PUBLIC bplcgcd_gf2)
add_library(bplcgcd_channel STATIC src/channel/channel.c)
target_link_libraries(bplcgcd_channel PUBLIC bplcgcd_gf2)
add_library(bplcgcd_quantum STATIC src/core/quantum.c src/channel/quantum_channel.c src/decoders/qbp.c src/decoders/quantum_soft.c src/decoders/quantum_decoder.c)
target_link_libraries(bplcgcd_quantum PUBLIC bplcgcd_gf2 bplcgcd_search)
target_link_libraries(bp_lcgcd PRIVATE bplcgcd_quantum)
if(NOT MSVC)
    target_link_libraries(bplcgcd_channel PUBLIC m)
    target_link_libraries(bplcgcd_quantum PUBLIC m)
endif()
add_library(bplcgcd_simulation STATIC src/simulation/simulation.c)
target_link_libraries(bplcgcd_simulation PUBLIC bplcgcd_channel bplcgcd_decoder)

foreach(target IN ITEMS bp_lcgcd bplcgcd_gf2 bplcgcd_decoder bplcgcd_search bplcgcd_channel bplcgcd_simulation bplcgcd_quantum)
    set_target_properties(${target} PROPERTIES
        C_STANDARD 11
        C_STANDARD_REQUIRED YES
        C_EXTENSIONS NO
    )
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

include(CTest)
if(BUILD_TESTING)
    foreach(test IN ITEMS gf2 systematic decoder tep gcd lcgcd lcgcd_oracle channel simulation simulation_pair local_constraints trellis viterbi slva quantum qbp quantum_decoder)
        add_executable(${test}_test tests/${test}_test.c)
        if(test STREQUAL "quantum" OR test STREQUAL "qbp" OR test STREQUAL "quantum_decoder")
            target_link_libraries(${test}_test PRIVATE bplcgcd_quantum)
        elseif(test STREQUAL "simulation" OR test STREQUAL "simulation_pair")
            target_link_libraries(${test}_test PRIVATE bplcgcd_simulation)
        elseif(test STREQUAL "channel")
            target_link_libraries(${test}_test PRIVATE bplcgcd_channel bplcgcd_decoder)
        elseif(test STREQUAL "decoder" OR test STREQUAL "gcd" OR test STREQUAL "lcgcd" OR test STREQUAL "lcgcd_oracle" OR test STREQUAL "local_constraints" OR test STREQUAL "trellis")
            target_link_libraries(${test}_test PRIVATE bplcgcd_decoder)
        elseif(test STREQUAL "tep" OR test STREQUAL "viterbi" OR test STREQUAL "slva")
            target_link_libraries(${test}_test PRIVATE bplcgcd_search)
        else()
            target_link_libraries(${test}_test PRIVATE bplcgcd_gf2)
        endif()
        set_target_properties(${test}_test PROPERTIES
            C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
        if((test STREQUAL "qbp" OR test STREQUAL "quantum_decoder") AND UNIX AND NOT APPLE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_definitions(${test}_test PRIVATE QBP_TRACK_ALLOCATIONS)
            target_link_options(${test}_test PRIVATE
                -Wl,--wrap=malloc -Wl,--wrap=calloc -Wl,--wrap=realloc -Wl,--wrap=free)
            if(test STREQUAL "quantum_decoder")
                target_link_options(quantum_decoder_test PRIVATE -Wl,--wrap=binary_matrix_systematize)
            endif()
        endif()
        if(MSVC)
            target_compile_options(${test}_test PRIVATE /W4)
        else()
            target_compile_options(${test}_test PRIVATE -Wall -Wextra -Wpedantic)
        endif()
        if(test STREQUAL "lcgcd_oracle")
            add_test(NAME ${test} COMMAND ${test}_test "${CMAKE_CURRENT_BINARY_DIR}/lcgcd_oracle_debug.log")
        elseif(test STREQUAL "simulation_pair")
            add_test(NAME ${test} COMMAND ${test}_test "${CMAKE_CURRENT_BINARY_DIR}/simulation_pair_debug.log")
        elseif(test STREQUAL "quantum_decoder")
            add_test(NAME ${test} COMMAND ${test}_test "${CMAKE_CURRENT_BINARY_DIR}/quantum_decoder_debug.log")
        else()
            add_test(NAME ${test} COMMAND ${test}_test)
        endif()
        set_tests_properties(${test} PROPERTIES TIMEOUT 10)
    endforeach()
    set_tests_properties(lcgcd_oracle PROPERTIES TIMEOUT 30)
    set_tests_properties(quantum_decoder PROPERTIES TIMEOUT 30)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME quantum_sim_cli COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/quantum_sim_cli.py" "$<TARGET_FILE:bp_lcgcd>"
            "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/quantum-sim")
        set_tests_properties(quantum_sim_cli PROPERTIES TIMEOUT 30)
    endif()
endif()
